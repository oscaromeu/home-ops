% Regenerate: nix-shell -p texliveSmall poppler_utils --run "pdflatex gitops-workflow.tex && pdftocairo -svg gitops-workflow.pdf gitops-workflow.svg"
% lmodern matters: without it pdflatex emits bitmap fonts and the SVG text comes out rasterised.
\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, calc}

% Never black!N outside a filled box: it vanishes on dark backgrounds. This grey
% is the value that maximises the worst of the two contrasts, white and #0d1117.
\definecolor{ink}{HTML}{6E7681}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily\LARGE,
  node distance=14mm and 16mm,
  box/.style={
    rounded corners=3pt, draw, line width=0.9pt,
    align=center, inner sep=9pt, minimum width=50mm, minimum height=12mm
  },
  file/.style ={box, fill=gray!20,   draw=gray!70!black},
  rset/.style ={box, fill=blue!18,   draw=blue!60!black},
  ks/.style   ={box, fill=green!20,  draw=green!55!black},
  helm/.style ={box, fill=orange!25, draw=orange!75!black},
  flow/.style ={-{Stealth[length=2.6mm]}, draw=ink, line width=0.9pt},
  dep/.style  ={-{Stealth[length=2.6mm]}, draw=ink, line width=0.9pt, dashed},
  lbl/.style  ={font=\sffamily\large, text=ink, inner sep=4pt},
  note/.style ={font=\sffamily\large, text=ink, inner sep=4pt}
]

\node[file] (a)                        {\textbf{inputs/default.yaml}};
\node[rset, right=of a] (b)            {\texttt{rs/}\textbf{apps-default}};

\node[ks, above right=2mm and 22mm of b] (n) {\texttt{ns/}\textbf{default}};
\node[ks, below right=2mm and 22mm of b] (c) {\texttt{ks/}\textbf{navidrome}};

\node[helm, right=32mm of c] (e)       {\texttt{hr/}\textbf{navidrome}};
\node[ks, below=24mm of c] (g)         {\texttt{ks/}\textbf{onepassword}};

\draw[flow] (a) -- node[lbl, above] {becomes} (b);
\draw[flow] (b) -- node[lbl, above left=-1mm] {generates} (n);
\draw[flow] (b) -- node[lbl, below left=-1mm, align=center] {generates\\one per app} (c);
\draw[flow] (c) -- node[lbl, above] {creates} (e);
\draw[dep]  (c) -- node[lbl, right] {depends on} (g);

% The one thing a reader cannot guess: everything right of this line is
% created at reconcile time and is written down nowhere.
\coordinate (cstack) at ($(c.north east)+(2.2mm,2.2mm)$);

% Stacked cards: one app is drawn, but a ResourceSet generates one of these per
% app. Drawn as copies of the real node rectangle, so they cannot drift in size.
% The namespace deliberately has no stack - there is exactly one of those.
\begin{scope}[on background layer]
  \foreach \d in {1.1mm, 2.2mm} {
    \draw[fill=green!20, draw=green!55!black, line width=0.9pt, rounded corners=3pt]
      ($(c.south west)+(\d,\d)$) rectangle ($(c.north east)+(\d,\d)$);
    \draw[fill=orange!25, draw=orange!75!black, line width=0.9pt, rounded corners=3pt]
      ($(e.south west)+(\d,\d)$) rectangle ($(e.north east)+(\d,\d)$);
  }
\end{scope}

\begin{scope}[on background layer]
  \coordinate (mid) at ($(b.east)!0.5!(c.west)$);
  \draw[ink, dotted, line width=1pt] ($(mid)+(0,3.3)$) -- ($(mid)+(0,-5.2)$);
  % Column headers: each half is named above the content it describes, and the
  % divider runs the full height. A second dashed box would collide with the
  % region already inside the right-hand side.
  \coordinate (axis) at (0,3.6);
  % |<->|<->| across the full width: a bar at each end and one at the split.
  \draw[ink, |<->|, line width=0.7pt] (a.west |- axis) -- (mid |- axis);
  \draw[ink, |<->|, line width=0.7pt] (mid |- axis) -- (e.east |- axis);
  \node[note, above=1mm] at ($(a.west |- axis)!0.5!(mid |- axis)$) {written in git};
  \node[note, above=1mm] at ($(mid |- axis)!0.5!(e.east |- axis)$) {created at reconcile time};
\end{scope}

% The region is what mermaid cannot draw: it marks where one ResourceSet's
% output ends. onepassword sits outside it because another one generates it.
\begin{scope}[on background layer]
  \node[draw=ink, dashed, rounded corners=5pt,
        inner sep=7mm, fit=(n)(c)(cstack)] (box) {};
  \node[note, anchor=north west, font=\sffamily\normalsize] at ($(box.north west)+(2.5mm,-2mm)$) {generated by \texttt{apps-default}};
  \node[note, below=3mm of g.south] {generated by \texttt{apps-external-secrets}};
\end{scope}

\node[draw=ink, rounded corners=3pt, line width=0.7pt, inner sep=3mm,
      anchor=north west, align=left, font=\sffamily\large, text=ink]
  at ($(a.west)+(0,-2.2)$)
  {\begin{tabular}{@{}l@{~~}l@{}}
     \texttt{rs/} & ResourceSet\\
     \texttt{ns/} & Namespace\\
     \texttt{ks/} & Kustomization\\
     \texttt{hr/} & HelmRelease
   \end{tabular}};

\end{tikzpicture}
\end{document}
