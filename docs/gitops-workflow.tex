\documentclass[border=10pt]{standalone}

\usepackage[T1]{fontenc}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, calc}

% Never black!N outside a filled box: it vanishes on dark backgrounds. This grey
% is the value that maximises the worst of the two contrasts, white and #0d1117.
\definecolor{ink}{HTML}{6E7681}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily\LARGE,
  node distance=14mm and 16mm,
  box/.style={
    rounded corners=3pt, draw, line width=0.9pt,
    align=center, inner sep=9pt, minimum width=50mm, minimum height=12mm
  },
  file/.style ={box, fill=gray!20,   draw=gray!70!black},
  cfg/.style  ={box, fill=blue!18,   draw=blue!60!black},
  ks/.style   ={box, fill=green!20,  draw=green!55!black},
  helm/.style ={box, fill=orange!25, draw=orange!75!black},
  flow/.style ={-{Stealth[length=2.6mm]}, draw=ink, line width=0.9pt},
  dep/.style  ={-{Stealth[length=2.6mm]}, draw=ink, line width=0.9pt, dashed},
  lbl/.style  ={font=\sffamily\large, text=ink, inner sep=4pt},
  note/.style ={font=\sffamily\large, text=ink, inner sep=4pt}
]

\node[file] (a)                        {\textbf{apps/default/navidrome/ks.yaml}};
\node[ks, right=30mm of a] (c)         {\texttt{ks/}\textbf{navidrome}};
\node[helm, right=32mm of c] (e)       {\texttt{hr/}\textbf{navidrome}};
\node[ks, below=24mm of c] (g)         {\texttt{ks/}\textbf{onepassword}};
\node[cfg, above=20mm of c] (p)        {\textbf{kustomization.yaml}};

\draw[flow] (a) -- node[lbl, above] {becomes} (c);
\draw[flow] (c) -- node[lbl, above] {creates} (e);
\draw[dep]  (c) -- node[lbl, right] {depends on} (g);
\draw[dep]  (p) -- node[lbl, left, align=right, pos=0.3] {injects the\\shared defaults} (c);

% Stacked cards: one app is drawn, but the cluster carries one file and one
% Kustomization per app. Drawn as copies of the real node rectangle, so they
% cannot drift in size.
\begin{scope}[on background layer]
  \foreach \k in {2,1} {
    \draw[fill=gray!20, draw=gray!70!black, line width=0.9pt, rounded corners=3pt]
      ($(a.south west)+(\k*2.2mm,\k*4.5mm)$) rectangle ($(a.north east)+(\k*2.2mm,\k*4.5mm)$);
    \draw[fill=green!20, draw=green!55!black, line width=0.9pt, rounded corners=3pt]
      ($(c.south west)+(\k*2.2mm,\k*4.5mm)$) rectangle ($(c.north east)+(\k*2.2mm,\k*4.5mm)$);
    \draw[fill=orange!25, draw=orange!75!black, line width=0.9pt, rounded corners=3pt]
      ($(e.south west)+(\k*2.2mm,\k*4.5mm)$) rectangle ($(e.north east)+(\k*2.2mm,\k*4.5mm)$);
  }
\end{scope}

\node[note, below=3mm of g.south] {declared in \texttt{apps/external-secrets/}};
\node[note, above=3mm of p.north] {the cluster root, patching every \texttt{ks}};

\node[draw=ink, rounded corners=3pt, line width=0.7pt, inner sep=3mm,
      anchor=north west, align=left, font=\sffamily\large, text=ink]
  at ($(a.west)+(0,-2.2)$)
  {\begin{tabular}{@{}l@{~~}l@{}}
     \texttt{ks/} & Kustomization\\
     \texttt{hr/} & HelmRelease
   \end{tabular}};
